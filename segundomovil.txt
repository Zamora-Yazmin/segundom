ADAPTADOR.JAVA
package com.example.examenreciclado;

import android.content.Context;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.BaseAdapter;
import android.widget.ImageView;
import android.widget.TextView;

public class Adaptador extends BaseAdapter {
    Context contexto;
    String listaositos[];
    String listaComentar[];
    int listaImagenes[];
    LayoutInflater inflater;

    public Adaptador(Context contexto, String[] listaositos, String[] listaComentar, int[] listaImagenes) {
        this.contexto = contexto;
        this.listaositos = listaositos;
        this.listaComentar = listaComentar;
        this.listaImagenes = listaImagenes;
        this.inflater = LayoutInflater.from(contexto);
    }

    @Override
    public int getCount() { return listaComentar.length; }

    @Override
    public Object getItem(int position) { return null; }

    @Override
    public long getItemId(int position) { return 0; }

    @Override
    public View getView(int position, View convertView, ViewGroup parent) {
        convertView = inflater.inflate(R.layout.basurita, null);
        TextView textView = convertView.findViewById(R.id.textview);
        TextView textView2 = convertView.findViewById(R.id.textview2);
        ImageView ositoimg = convertView.findViewById(R.id.imageicon);

        textView.setText(listaositos[position]);
        textView2.setText(listaComentar[position]);
        ositoimg.setImageResource(listaImagenes[position]);

        return convertView;
    }
}

MAIN
package com.example.examenreciclado;

import android.content.Intent;
import android.os.Bundle;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ListView;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity implements AdapterView.OnItemClickListener {

    String ositosComenta[] = {"Ir al Scroll Horizontal", "Ir al Menú de Opciones"};
    String ositosListe[] = {"01", "02"};
    int ositosImagenes[] = {R.drawable.amarillo, R.drawable.amorosita};

    ListView listView;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        listView = findViewById(R.id.vistitas);
        listView.setOnItemClickListener(this);

        Adaptador adaptadorcito = new Adaptador(this, ositosListe, ositosComenta, ositosImagenes);
        listView.setAdapter(adaptadorcito);
    }

    @Override
    public void onItemClick(AdapterView<?> parent, View view, int position, long id) {
        if (position == 0) {
            // Envía a la segunda actividad (HorizontalScrollView)
            Intent intent = new Intent(this, SegundaActivity.class);
            startActivity(intent);
        } else if (position == 1) {
            // Envía a la tercera actividad (Menú de opciones)
            Intent intent2 = new Intent(this, TerceraActivity.class);
            startActivity(intent2);
        }
    }
}

package com.example.examenreciclado;

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.widget.ImageView;
import android.widget.LinearLayout;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;

public class SegundaActivity extends AppCompatActivity {

    String ositosComenta[] = {"uno", "dos", "tres", "cuatro", "cinco", "seis", "siete", "ocho", "nueve", "diez"};
    String ositosListe[] = {"01", "02", "03", "04", "05", "06", "07", "08", "09", "010"};
    int ositosImagenes[] = {
        R.drawable.amarillo, R.drawable.amorosita, R.drawable.cafe,
        R.drawable.amorosita, R.drawable.amarillo, R.drawable.cafe,
        R.drawable.amorosita, R.drawable.amarillo, R.drawable.cafe, R.drawable.amorosita
    };

    LinearLayout contenedorHorizontal;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_segunda);

        contenedorHorizontal = findViewById(R.id.visititas);
        LayoutInflater inflater = LayoutInflater.from(this);

        for (int i = 0; i < ositosComenta.length; i++) {
            View itemView = inflater.inflate(R.layout.basurita, contenedorHorizontal, false);
            
            TextView textView = itemView.findViewById(R.id.textview);
            TextView textView2 = itemView.findViewById(R.id.textview2);
            ImageView ositoimg = itemView.findViewById(R.id.imageicon);

            textView.setText(ositosComenta[i]);
            textView2.setText(ositosListe[i]);
            ositoimg.setImageResource(ositosImagenes[i]);

            contenedorHorizontal.addView(itemView);
        }
    }
}
